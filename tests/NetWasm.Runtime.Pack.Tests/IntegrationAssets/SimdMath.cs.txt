using System.Runtime.InteropServices.JavaScript;
using System.Runtime.Intrinsics;

public static class MathProject
{
    [JSExport]
    public static int Run(int input)
    {
        var a = Vector128.Create(input, input + 1, input - 3, 5);
        var b = a * Vector128.Create(3) + Vector128.Create(7);
        var f = Vector128.Create(input * 0.25f, 9f, 2.5f, -3f);
        var root = Vector128.Sqrt(Vector128.Abs(f));
        return b.GetElement(0) + b.GetElement(1) + b.GetElement(2) + b.GetElement(3) +
            (int)(root.GetElement(0) + root.GetElement(1) + root.GetElement(2) + root.GetElement(3));
    }
}
